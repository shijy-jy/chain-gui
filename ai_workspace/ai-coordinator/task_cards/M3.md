### Step 3：写"选目录"UI + 调 invoke('scan_chain')

在 `src/routes/+page.svelte`（或主页面）：

```svelte
<script lang="ts">
  import { invoke } from '@tauri-apps/api/core';
  import { open } from '@tauri-apps/plugin-dialog';
  
  let chainDir = $state<string | null>(null);
  let snapshot = $state<any | null>(null);
  let error = $state<string | null>(null);
  let loading = $state(false);
  
  async function pickChainDir() {
    const selected = await open({ directory: true, multiple: false });
    if (typeof selected === 'string') {
      chainDir = selected;
      await loadChain();
    }
  }
  
  async function loadChain() {
    if (!chainDir) return;
    loading = true;
    error = null;
    try {
      snapshot = await invoke('scan_chain', { dir: chainDir });
    } catch (e) {
      error = String(e);
    } finally {
      loading = false;
    }
  }
</script>

<button onclick={pickChainDir} disabled={loading}>
  {loading ? '加载中...' : '选 .chain 父目录'}
</button>

{#if chainDir}
  <p>目录：{chainDir}</p>
{/if}

{#if error}
  <p style="color: red">错误：{error}</p>
{/if}

{#if snapshot}
  <p>节点数：{snapshot.manifest.node_count}，边数：{snapshot.manifest.edge_count}</p>
{/if}
```

```typescript
export type NodeType = 'goal' | 'design' | 'task' | 'verification';
export type NodeStatus = 'pending' | 'in_progress' | 'success' | 'failed' | 'blocked';

export interface ChainNode {
  id: string;
  type: NodeType;
  title: string;
  parent: string | null;
  status: NodeStatus;
  created: string;
  updated: string;
  revision: number;
  tags: string[];
  body: string;
}

export interface ChainEdge {
  parent: string;
  child: string;
}

export interface ChainManifest {
  root: string;
  node_count: number;
  edge_count: number;
  generated_at: string;
}

export interface ValidationReport {
  valid: boolean;
  errors: string[];
  warnings: string[];
}

export interface ChainSnapshot {
  nodes: ChainNode[];
  edges: ChainEdge[];
  manifest: ChainManifest;
  validation: ValidationReport;
}
```

### Step 5：写 ChainSnapshot → cytoscape elements 转换

新建 `src/lib/chain_to_cytoscape.ts`：

```typescript
import type { ElementDefinition } from 'cytoscape';
import type { ChainSnapshot } from './types';

export function chainToElements(snap: ChainSnapshot): ElementDefinition[] {
  const elements: ElementDefinition[] = [];
  
  for (const node of snap.nodes) {
    elements.push({
      data: {
        id: node.id,
        label: `${node.id}\n${node.title}`,
        nodeType: node.type,
        nodeStatus: node.status,
        parent: node.parent,
      },
    });
  }
  
  for (const edge of snap.edges) {
    elements.push({
      data: {
        id: `${edge.parent}->${edge.child}`,
        source: edge.parent,
        target: edge.child,
      },
    });
  }
  
  return elements;
}
```

```svelte
<script lang="ts">
  import cytoscape from 'cytoscape';
  import dagre from 'cytoscape-dagre';
  import type { Stylesheet } from 'cytoscape';
  import { onMount, onDestroy } from 'svelte';
  import { chainToElements } from '$lib/chain_to_cytoscape';
  import type { ChainSnapshot } from '$lib/types';
  
  cytoscape.use(dagre);
  
  // ... 前面 Step 3 的代码保留 ...
  
  let container: HTMLDivElement;
  let cy: cytoscape.Core | null = null;
  
  // 节点配色 + 边样式
  const style: Stylesheet[] = [
    {
      selector: 'node',
      style: {
        'label': 'data(label)',
        'text-wrap': 'wrap',
        'text-valign': 'center',
        'text-halign': 'center',
        'width': 120,
        'height': 60,
        'background-color': '#eee',
        'border-width': 3,
        'border-color': '#999',
        'font-size': '11px',
      },
    },
    // NodeType 配色（背景色）
    { selector: 'node[nodeType = "goal"]',        style: { 'background-color': '#a8e6a3' } },
    { selector: 'node[nodeType = "design"]',      style: { 'background-color': '#a3c9e6' } },
    { selector: 'node[nodeType = "task"]',        style: { 'background-color': '#f5b97c' } },
    { selector: 'node[nodeType = "verification"]',style: { 'background-color': '#c9a3e6' } },
    // NodeStatus 边框
    { selector: 'node[nodeStatus = "pending"]',     style: { 'border-color': '#999' } },
    { selector: 'node[nodeStatus = "in_progress"]', style: { 'border-color': '#f5d76e', 'border-width': 5 } },
    { selector: 'node[nodeStatus = "success"]',     style: { 'border-color': '#4caf50' } },
    { selector: 'node[nodeStatus = "failed"]',      style: { 'border-color': '#f44336' } },
    { selector: 'node[nodeStatus = "blocked"]',     style: { 'border-color': '#666' } },
    // 边
    {
      selector: 'edge',
      style: {
        'width': 2,
        'line-color': '#666',
        'target-arrow-color': '#666',
        'target-arrow-shape': 'triangle',
        'curve-style': 'bezier',
      },
    },
  ];
  
  $effect(() => {
    if (cy && snapshot) {
      cy.elements().remove();
      cy.add(chainToElements(snapshot));
      cy.layout({ name: 'dagre', rankDir: 'LR', nodeSep: 50, rankSep: 100 }).run();
    }
  });
  
  onMount(() => {
    cy = cytoscape({ container, style, elements: [], layout: { name: 'dagre', rankDir: 'LR' } });
  });
  
  onDestroy(() => cy?.destroy());
</script>
```
