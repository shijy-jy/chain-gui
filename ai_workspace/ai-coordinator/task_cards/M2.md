```rust
use anyhow::{Result, Context};
use crate::model::node::Node;

pub fn parse_node_file(content: &str) -> Result<(Node, String)> {
    // 节点文件格式：
    // ---
    // id: t-001
    // type: task
    // ...
    // ---
    //
    // # Markdown 正文...
    
    let content = content.trim_start();
    if !content.starts_with("---") {
        anyhow::bail!("节点文件必须以 --- 开头的 frontmatter 起始");
    }
    let rest = &content[3..];
    let end = rest.find("\n---").context("frontmatter 未闭合（找不到 ---）")?;
    let yaml_str = &rest[..end];
    let body = rest[end + 4..].trim_start_matches('\n').to_string();
    
    let node: Node = serde_yaml::from_str(yaml_str)
        .context("frontmatter YAML 解析失败")?;
    Ok((node, body))
}

#[cfg(test)]
mod tests {
    use super::*;

id: t-001
type: task

parent: d-001
status: pending
created: 2026-08-13T10:00:00+08:00
updated: 2026-08-13T10:00:00+08:00
revision: 1
tags: [test]
---

# 测试节点正文

正文内容...
"#;
        let (node, body) = parse_node_file(content).unwrap();
```

**`walker.rs`**（递归扫描 `dir/.chain/nodes/*.md`）：

```rust
use std::path::{Path, PathBuf};
use anyhow::{Result, Context};
use walkdir::WalkDir;
use crate::scanner::frontmatter::parse_node_file;
use crate::model::chain::{ChainSnapshot, Manifest, Edge};

pub fn scan_chain_dir(root: &Path) -> Result<ChainSnapshot> {
    let chain_dir = root.join(".chain");
    let nodes_dir = chain_dir.join("nodes");
    
    if !chain_dir.exists() {
        anyhow::bail!("目录 {:?} 下不存在 .chain/ 子目录", root);
    }
    
    let mut nodes = Vec::new();
    let mut edges = Vec::new();
    
    for entry in WalkDir::new(&nodes_dir)
        .follow_links(false)
        .into_iter()
        .filter_map(|e| e.ok())
    {
        let path = entry.path();
        if path.extension().and_then(|s| s.to_str()) != Some("md") {
            continue;
        }
        let content = std::fs::read_to_string(path)
            .with_context(|| format!("读取节点文件 {:?} 失败", path))?;
        let (node, _body) = parse_node_file(&content)
            .with_context(|| format!("解析节点文件 {:?} 失败", path))?;
        
        if let Some(parent_id) = &node.parent {
            edges.push(Edge {
                parent: parent_id.clone(),
                child: node.id.clone(),
            });
        }
        nodes.push(node);
    }
    
    let manifest = Manifest {
        root: root.to_path_buf(),
        node_count: nodes.len(),
        edge_count: edges.len(),
        generated_at: chrono_now_iso8601(),
    };
    
    Ok(ChainSnapshot {
        nodes,
        edges,
        manifest,
        validation: Default::default(),
    })
}

fn chrono_now_iso8601() -> String {
    use std::time::{SystemTime, UNIX_EPOCH};
    let secs = SystemTime::now()
        .duration_since(UNIX_EPOCH)
        .map(|d| d.as_secs())
        .unwrap_or(0);
    format!("1970-01-01T00:00:00+00:00 +{}s", secs)  // 简化版，M3 阶段用 chrono 替换
}
```

### Step 3：暴露 Tauri command `scan_chain`

在 `src-tauri/src/lib.rs`（或新建 `src-tauri/src/commands/scan.rs`）：

```rust
use tauri::command;
use std::path::PathBuf;
use crate::scanner::walker::scan_chain_dir;
use crate::model::chain::ChainSnapshot;

#[command]
pub fn scan_chain(dir: String) -> Result<ChainSnapshot, String> {
    let path = PathBuf::from(&dir);
    scan_chain_dir(&path).map_err(|e| e.to_string())
}

#[cfg_attr(mobile, tauri::mobile_entry_point)]
pub fn run() {
    tauri::Builder::default()
        .invoke_handler(tauri::generate_handler![scan_chain])
        .run(tauri::generate_context!())
        .expect("error while running tauri application");
}
```

```rust
use chain_gui_lib::scanner::walker::scan_chain_dir;
use std::fs;
use tempfile::TempDir;

#[test]
fn test_scan_real_chain() {
    let tmp = TempDir::new().unwrap();
    let root = tmp.path();
    let chain_dir = root.join(".chain");
    let nodes_dir = chain_dir.join("nodes");
    fs::create_dir_all(&nodes_dir).unwrap();
    
    // 写 3 个测试节点

        nodes_dir.join("g-001.md"),
        "---\nid: g-001\ntype: goal\ntitle: 顶层目标\nparent: \nstatus: pending\ncreated: 2026-08-13T10:00:00+08:00\nupdated: 2026-08-13T10:00:00+08:00\nrevision: 1\ntags: []\n---\n\n# 顶层目标\n",
    ).unwrap();
    
    fs::write(
        nodes_dir.join("d-001.md"),
        "---\nid: d-001\ntype: design\ntitle: 设计1\nparent: g-001\nstatus: in_progress\ncreated: 2026-08-13T10:00:00+08:00\nupdated: 2026-08-13T10:00:00+08:00\nrevision: 1\ntags: []\n---\n\n# 设计1\n",

    fs::write(
        nodes_dir.join("t-001.md"),
```
